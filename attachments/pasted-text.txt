2.2. Dynamic Loader (router/skill_loader.py)
# router/skill_loader.py -------------------------------------------------
import importlib
import pathlib
import yaml
from typing import Tuple, Dict, Any

REGISTRY_ROOT = pathlib.Path(__file__).parents[1] / "solanabr" / "skills"

class SkillLoadError(RuntimeError): pass
class PolicyViolationError(RuntimeError): pass

def _read_descriptor(skill_dir: pathlib.Path) -> Dict[str, Any]:
    desc_path = skill_dir / "skill.yaml"
    if not desc_path.is_file():
        raise SkillLoadError(f"Missing skill.yaml in {skill_dir}")
    return yaml.safe_load(desc_path.read_text())

def _verify_mandatory_gates(descriptor: Dict[str, Any]) -> None:
    required = {
        "no_secrets",
        "no_implicit_tx",
        "pin_source",
        "treat_untrusted",
        "progressive_load",
        "require_evidence",
        "verify_existence",
    }
    missing = required - set(descriptor.get("mandatory_gates", []))
    if missing:
        raise PolicyViolationError(
            f"Skill {descriptor.get('name')} lacks mandatory gates {missing}"
        )

def load_skill(name: str) -> Tuple[Any, Dict[str, Any]]:
    """
    Dynamically import the skill module and return (module, descriptor).
    Raises SkillLoadError / PolicyViolationError on failure.
    """
    for skill_dir in REGISTRY_ROOT.iterdir():
        if not skill_dir.is_dir():
            continue
        desc = _read_descriptor(skill_dir)
        if desc.get("name") == name:
            _verify_mandatory_gates(desc)
            entrypoint = desc.get("entrypoint", "skill.py")
            module_name = f"solanabr.skills.{skill_dir.name}.{entrypoint[:-3]}"
            try:
                mod = importlib.import_module(module_name)
            except Exception as exc:
                raise SkillLoadError(
                    f"Import of {module_name} failed: {exc}"
                ) from exc
            return mod, desc
    raise SkillLoadError(f"Skill {name!r} not found in registry")
# -------------------------------------------------------------------------
Only the primary skill and the explicitly‑listed secondary skills are loaded. All others stay on disk, never imported.

3.2. Rule Engine (router/rule_engine.py)
# router/rule_engine.py -------------------------------------------------------
import yaml
import fnmatch
from typing import List, Dict, Any

class RuleEngine:
    def __init__(self, path: str = "router/rules.yaml"):
        self.rules: List[Dict[str, Any]] = yaml.safe_load(open(path))

    def match(self, classification: Dict[str, Any]) -> List[Dict[str, Any]]:
        """
        Return a list of matching rules ordered by descending priority.
        """
        matches = [r for r in self.rules if self._matches(r, classification)]
        matches.sort(key=lambda r: r.get("priority", 0), reverse=True)
        return matches

    @staticmethod
    def _matches(rule: Dict[str, Any], cls: Dict[str, Any]) -> bool:
        # intent: exact match
        if rule.get("intent") and cls.get("intent") != rule["intent"]:
            return False

        # domain: any rule domain matches any request domain
        if rule.get("domains"):
            if not any(
                any(fnmatch.fnmatch(d, pattern) for pattern in rule["domains"])
                for d in cls.get("domains", [])
            ):
                return False

        # language: same idea
        if rule.get("languages"):
            if not any(
                any(fnmatch.fnmatch(l, pat) for pat in rule["languages"])
                for l in cls.get("languages", [])
            ):
                return False

        return True
# -------------------------------------------------------------------------

# router/classifier.py -------------------------------------------------------
import re
from typing import Dict, List

def classify(raw: str) -> Dict[str, any]:
    # Intent
    intent = "generic"
    if re.search(r"\baudit\b", raw, re.I):
        intent = "security_audit"
    elif re.search(r"\bimplement\b", raw, re.I):
        intent = "implementation"
    elif re.search(r"\binspect\b|check\b", raw, re.I):
        intent = "analysis"

    # Domains (simple keyword map – extend as you add new domains)
    domain_map = {
        "anchor": "anchor",
        "token-2022": "token-2022",
        "spl-token": "spl-token",
        "simd": "simd",
        "poh": "consensus",
        "tower bft": "consensus",
        "agave": "runtime",
        "jupiter": "dex",
    }
    domains: List[str] = [
        d for kw, d in domain_map.items() if re.search(kw, raw, re.I)
    ]

    # Languages
    langs = []
    if re.search(r"\brust\b", raw, re.I):
        langs.append("rust")
    if re.search(r"\btypescript\b|\bjs\b|\bjavascript\b", raw, re.I):
        langs.append("typescript")

    return {
        "intent": intent,
        "domains": domains,
        "languages": langs,
    }

5. Mandatory‑Policy Gate – Cross‑cutting safety controls
All rules carry a gates list. The gate runner checks the runtime context that the router builds before any skill code runs.

# router/policy_gate.py ------------------------------------------------------
MANDATORY = {
    "no_secrets": lambda ctx: not ctx.get("has_secrets", False),
    "no_implicit_tx": lambda ctx: not ctx.get("implicit_tx", False),
    "pin_source": lambda ctx: ctx.get("source_pinned", False),
    "treat_untrusted": lambda ctx: not ctx.get("trusted_data", False),
    "progressive_load": lambda ctx: ctx.get("progressive_loaded", False),
    "require_evidence": lambda ctx: bool(ctx.get("evidence", [])),
    "verify_existence": lambda ctx: ctx.get("all_paths_exist", False),
}

def enforce(workflow: dict, ctx: dict) -> None:
    """
    Raise RuntimeError if any declared gate fails.
    """
    for gate in workflow.get("gates", []):
        checker = MANDATORY.get(gate)
        if checker is None:
            raise RuntimeError(f"Unknown gate {gate}")
        if not checker(ctx):
            raise RuntimeError(
                f"Gate {gate} failed – see policy documentation"
            )
# -------------------------------------------------------------------------
Constructing the context (called right after the rule engine decides on the route):

def build_policy_ctx(workflow):
    # 1️⃣ Evidence existence
    evidence_ok = bool(workflow.get("evidence"))
    # 2️⃣ Curriculum existence
    cur_paths = workflow.get("curriculum", [])
    paths_exist = all(pathlib.Path(p).exists() for p in cur_paths)
    # 3️⃣ Progressive load – true because we load *only* the skills we need
    # 4️⃣ No secrets / implicit tx – we keep these flags false in the core engine
    return {
        "has_secrets": False,
        "implicit_tx": False,
        "source_pinned": evidence_ok,
        "trusted_data": False,
        "progressive_loaded": True,
        "evidence": workflow.get("evidence", []),
        "all_paths_exist": paths_exist,
    }
If any gate fails the router aborts before any skill code is executed, guaranteeing safety.

6. Curriculum & Evidence Adapter – Concrete, reproducible, lock‑file based
6.1. Lock‑file (solana_curriculum.lock)
# format: <repo_url>|<local_path>|<commit_sha>|<tree_hash>|<fallback_flag>
https://github.com/solana-labs/anchor.git|repos/anchor|a1b2c3d4e5f6|e1f2d3c4b5a6|false
https://github.com/solana-labs/agave.git|repos/agave|7f8e9d0c|a5b6c7d8e9|false
Generated by router/generate_lockfile.py (see Section 7).

When the router needs a repo for evidence it never uses git pull; it checks the lock‑file, git checkout <commit> and verifies the tree hash. If the lock‑file entry has fallback:true, the router aborts unless the user passed --allow-fallback.

6.2. Curriculum verification (router/curriculum_adapter.py)
# router/curriculum_adapter.py ---------------------------------------------
import pathlib
from typing import List, Tuple

def resolve_paths(paths: List[str]) -> List[Tuple[str, str]]:
    """
    Returns a list of (path, status) where status ∈ {VERIFIED, DRAFT, MISSING}.
    """
    out = []
    for p in paths:
        real_path = pathlib.Path(p).expanduser().resolve()
        if not real_path.exists():
            out.append((str(real_path), "MISSING"))
        elif "draft" in real_path.name.lower():
            out.append((str(real_path), "DRAFT"))
        else:
            out.append((str(real_path), "VERIFIED"))
    return out
# -------------------------------------------------------------------------
The router calls this adapter after rule‑engine matching; the returned status is stored in the evidence.log.

get_tree_hash(repo_path, commit) is a tiny helper that runs:

git -C <repo_path> checkout <commit> \
 && git ls-tree -r <commit> | sha256sum | cut -d' ' -f1
All evidence is pinned to a specific commit and to a deterministic tree hash, guaranteeing reproducibility.

#!/usr/bin/env bash
set -euo pipefail
LOCKFILE="${PWD}/solana_curriculum.lock"

# Helper to read a line from the lockfile
lookup_repo() {
    local url=$1
    grep -F "${url}" "$LOCKFILE" | head -n1 | cut -d'|' -f2-4
}

clone_or_checkout() {
    local url=$1 dest=$2 expected_sha=$3 fallback_allowed=$4

    if [[ ! -d "$dest/.git" ]]; then
        echo "[INFO] Cloning $url → $dest"
        git clone "$url" "$dest"
    fi
    pushd "$dest" >/dev/null
    git fetch --all

    if git rev-parse --verify "$expected_sha" >/dev/null 2>&1; then
        git checkout "$expected_sha"
        echo "[OK] Checked out exact revision $expected_sha"
    else
        if [[ "$fallback_allowed" != "true" ]]; then
            echo "[ERROR] Expected commit $expected_sha missing and fallback disabled" >&2
            exit 1
        fi
        echo "[WARN] Commit $expected_sha not found – staying on current $(git rev-parse --abbrev-ref HEAD)"
    fi
    popd >/dev/null
}

8.4. generate_lockfile.py (one‑off or --update-lock)
#!/usr/bin/env python3
"""
Generate solana_curriculum.lock – a deterministic list of repos+commits.
Usage: python generate_lockfile.py   # writes lockfile in cwd
"""
import pathlib, subprocess, json, sys

LOCKFILE = pathlib.Path("solana_curriculum.lock")
REPOS = [
    # (url, local_path)
    ("https://github.com/coral-xyz/anchor.git", "repos/anchor"),
    ("https://github.com/solana-labs/agave.git", "repos/agave"),
    ("https://github.com/solana-labs/token-list.git", "repos/token-list"),
]

def git_commit_hash(path):
    out = subprocess.check_output(["git", "-C", str(path), "rev-parse", "HEAD"])
    return out.decode().strip()

def git_tree_hash(path, commit):
    out = subprocess.check_output(
        ["git", "-C", str(path), "ls-tree", "-r", commit]
    )
    return subprocess.check_output(["sha256sum"], input=out).decode().split()[0]

def main():
    lines = []
    for url, local in REPOS:
        p = pathlib.Path(local)
        if not p.is_dir():
            print(f"[WARN] {local} not cloned – skipping", file=sys.stderr)
            continue
        commit = git_commit_hash(p)
        tree_hash = git_tree_hash(p, commit)
        lines.append(f"{url}|{local}|{commit}|{tree_hash}|false")
    LOCKFILE.write_text("\n".join(lines) + "\n")
    print(f"[INFO] Lockfile written to {LOCKFILE}")

if __name__ == "__main__":
    main()

9. End‑to‑End CLI (router_cli.py)
#!/usr/bin/env python3
# router_cli.py ---------------------------------------------------------
import argparse, json, sys
from pathlib import Path

from router.classifier import classify
from router.rule_engine import RuleEngine
from router.policy_gate import enforce
from router.skill_loader import load_skill
from router.evidence_tracker import EvidenceLog
from router.curriculum_adapter import resolve_paths

def main() -> None:
    parser = argparse.ArgumentParser(
        description="Solana Skill Decision Engine – deterministic routing + safety gates"
    )
    parser.add_argument("prompt", help="Free‑form user request")
    parser.add_argument("--json", action="store_true", help="Emit full JSON output")
    parser.add_argument("--dry-run", action="store_true", help="Only classify & route – do not load any skill")
    parser.add_argument("--allow-fallback", action="store_true", help="Accept repository fallback branches")
    args = parser.parse_args()

    # -----------------------------------------------------------------
    # 1️⃣ Classification
    # -----------------------------------------------------------------
    classification = classify(args.prompt)

    # -----------------------------------------------------------------
    # 2️⃣ Match rules
    # -----------------------------------------------------------------
    engine = RuleEngine()
    matches = engine.match(classification)
    if not matches:
        sys.exit("[ERROR] No routing rule matches the request")

    # Primary rule = highest priority
    primary_rule = matches[0]

    # -----------------------------------------------------------------
    # 3️⃣ Build a deterministic workflow descriptor
    # -----------------------------------------------------------------
    workflow = {
        "request": args.prompt,
        "classification": classification,
        "primary_rule": primary_rule,
        "secondary_rules": matches[1:],
        # copy shortcut lists for the next stages
        "gates": primary_rule.get("gates", []),
        "curriculum": primary_rule.get("curriculum", []),
        "evidence": primary_rule.get("evidence", []),
    }

    if args.dry_run:
        print(json.dumps(workflow, indent=2))
        return

    # -----------------------------------------------------------------
    # 4️⃣ Policy gate – enforce all mandatory controls
    # -----------------------------------------------------------------
    ctx = {
        "has_secrets": False,
        "implicit_tx": False,
        "source_pinned": bool(workflow["evidence"]),
        "trusted_data": False,
        "progressive_loaded": True,          # we load only needed modules
        "evidence": workflow["evidence"],
        "all_paths_exist": all(Path(p).exists() for p in workflow["curriculum"]),
    }
    enforce(workflow, ctx)

    # -----------------------------------------------------------------
    # 5️⃣ Load primary + secondary skills (dynamic import)
    # -----------------------------------------------------------------
    primary_mod, primary_meta = load_skill(primary_rule["primary_skill"])
    secondary_loaded = []
    for sec in primary_rule.get("secondary_skills", []):
        try:
            mod, meta = load_skill(sec)
            secondary_loaded.append({"module": mod, "meta": meta, "name": sec})
        except Exception as e:
            # Secondary failure is non‑fatal – just log a warning
            print(f"[WARN] Secondary skill {sec} could not be loaded: {e}", file=sys.stderr)

    # -----------------------------------------------------------------
    # 6️⃣ Evidence log
    # -----------------------------------------------------------------
    ev = EvidenceLog()
    ev.request = args.prompt
    ev.primary_skill = primary_meta["name"]
    ev.secondary_skills = [s["meta"]["name"] for s in secondary_loaded]
    ev.revisions = [
        {
            "repo": ev_entry["repo"],
            "commit": ev_entry["commit"],
            "tree_hash": ev_entry.get("tree_hash", "unknown"),
        }
        for ev_entry in workflow["evidence"]
    ]
    ev.curriculum = [{"path": p, "status": s} for p, s in resolve_paths(workflow["curriculum"])]
    ev.save()

    # -----------------------------------------------------------------
    # 7️⃣ Run the primary skill (the skill decides what to do)
    # -----------------------------------------------------------------
    # All skill modules must expose a `run(workflow, evidence_log)` entry point.
    # The call is wrapped in a try/except that prints a friendly JSON error.
    try:
        primary_mod.run(workflow, ev)   # <-- skill‑specific logic lives here
    except Exception as exc:
        err = {"error": "skill_failure", "message": str(exc), "workflow": workflow}
        print(json.dumps(err, indent=2), file=sys.stderr)
        sys.exit(1)

    # -----------------------------------------------------------------
    # 8️⃣ Final output
    # -----------------------------------------------------------------
    if args.json:
        out = {"workflow": workflow, "evidence": ev.to_dict()}
        print(json.dumps(out, indent=2))
    else:
        print(f"✅ Primary skill: {primary_meta['name']}")
        if secondary_loaded:
            secs = ", ".join(s["meta"]["name"] for s in secondary_loaded)
            print(f"🔧 Secondary skills loaded: {secs}")
        print(f"📚 Curriculum files: {', '.join(workflow['curriculum'])}")
        print("🗂 Evidence recorded in evidence.json")
        print("🛡 All mandatory gates passed – you may now safely invoke the skill.")
# -------------------------------------------------------------------------
if __name__ == "__main__":
    main()

4. Findings from your curriculum setup script
Your setup.sh contains useful routing conventions, but several behaviors need to be corrected before an autonomous agent should rely on it.

The current implementation is a tested starter router, not yet a complete discovery engine for every skill in the Solana ecosystem.
The next version should dynamically read the skill registry from solanabr/skills, check which skills are actually installed, parse each skill's description and metadata, and retrieve the relevant SKILL.md plus companion references. It should also map curriculum lessons and companion repositories to topics and prerequisites, instead of relying solely on keyword matches.
The essential design rule is: the model may recommend a route, but deterministic gates must validate it. Every decision should record the matched rule, the reason for selection, the files actually loaded, the source revision and the evidence needed to finish the task.
